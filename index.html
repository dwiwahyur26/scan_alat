async function gantiKamera(deviceId) { if (kontrolKamera) { kontrolKamera.stop(); } const elemenVideo = document.getElementById('video-preview'); let namaUser = document.getElementById("username").value; // PERBAIKAN UTAMA: Mengunci resolusi standar agar video tidak blank hitam di HP const opsiKoneksi = { deviceId: deviceId, modifiers: { constraints: { video: { deviceId: deviceId, facingMode: "environment", width: { ideal: 640 }, height: { ideal: 480 } }, audio: false } } }; document.getElementById("status-pesan").innerText = "Menghubungkan ke lensa kamera..."; try { // Menjalankan proses scanning dengan resolusi yang sudah dikunci aman kontrolKamera = await codeReader.decodeFromVideoDevice(opsiKoneksi, elemenVideo, (result, error) => { if (result) { console.log("Terbaca:", result.text); hentikanKameraLive(); document.getElementById("status-pesan").innerText = "Sedang mencatat data..."; fetch(`${URL_APPS_SCRIPT}?id_alat=${encodeURIComponent(result.text)}&user=${encodeURIComponent(namaUser)}`) .then(response => response.text()) .then(responseText => { document.getElementById("status-pesan").innerText = responseText; alert(responseText); document.getElementById("status-pesan").innerText = ""; }) .catch(err => { document.getElementById("status-pesan").innerText = "Gagal mengirim data ke Sheets!"; console.error(err); }); } // Memastikan status tulisan kembali normal saat kamera menyala if (elemenVideo.srcObject && document.getElementById("status-pesan").innerText === "Menghubungkan ke lensa kamera...") { document.getElementById("status-pesan").innerText = "Arahkan kamera ke Barcode/QR Code"; } if (error && !(error instanceof ZXing.NotFoundException)) { console.error("Eror Scanner:", error); } }); } catch (err) { document.getElementById("status-pesan").innerText = "Lensa ini sibuk/terkunci. Coba pilih kamera lain di menu!"; console.error(err); } }
